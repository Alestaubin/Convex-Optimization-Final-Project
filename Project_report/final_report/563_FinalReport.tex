\section{Introduction}
\subsection{Notation}%
  \label{sub:Notation}
  Throughout this paper, we will denote the \textit{proximal operator} of $f$ with parameter $\lambda$ by 
  \begin{equation}
    \begin{split}
      \prox_{\lambda f}(v) := \argmin_x \left\{f(x) + \frac{1}{2\lambda} \| x - v\|_2^2 \right\}
    \end{split}
  \end{equation}
  The \textit{indicator function} for a convex set $C$ will be defined as 
  \begin{equation}
    \begin{split}
      \delta_C (x) := 
      \begin{cases}
        0, &\text{ if } x \in C\\
        +\infty, &\text{ if } x \notin C
      \end{cases}
    \end{split}
  \end{equation}
  We also denote the \textit{Euclidean Projection} on a set $C$ by 
  \begin{equation}
    \begin{split}
      \proj_C(v):= \argmin_{x\in C} \| x -v \|_2
    \end{split}
  \end{equation}
  \subsection{Preleminaries}%
  \label{sub:Preleminaries}
  We first note an important relation between the proximal and subdifferential operators.\\
  \begin{proposition}
    Given a function $f\in\Gamma$, we have 
    \begin{equation}\label{subProx}
      \prox_{\lambda f} =(I+ \lambda \partial f)^{-1},
    \end{equation}  
    where $(I + \lambda \partial f)^{-1}$, for $\lambda > 0$, is called the \textbf{resolvent} of the operator $\partial f$. In particular, the proximal operator is the resolvent of the subdifferential operator \cite{parikh_2014}.\\
  \end{proposition}
  \begin{definition}[Maximal Monotone operator]
    A monotone operator is \textbf{maximal} if its graph is not properly contained in the graph of any other monotone operator. If A is monotone (resp. maximal monotone) then so are $A^{-1}$ and $\lambda A$ if $\lambda > 0$.
  \end{definition}
  The next proposition, given in \cite{ConnorVander}, will be useful in the construction of the optimizing algorithms.\\ 
  \begin{proposition}     
If $F$ is maximal monotone, then $(I + \lambda F)^{-1}(\hat x)$ exists and is unique for all $\hat x \in R^n$. The value $x = (I + \lambda F)^{-1}(\hat x)$ of the resolvent is the unique solution of the monotone inclusion
    \begin{equation}\label{maxmon}
      \hat x \in x + \lambda {F}(x) 
   \end{equation}
  \end{proposition}  
\begin{proposition}[Proximal operator splitting] \label{proxsplit}
  If $f$ is separable across two variables, so $f(x, y) = \varphi(x) + \psi(y)$, then
  \begin{equation}
    \prox_f (x,y) = (\prox_\varphi(x), \prox_\psi(y))
  \end{equation}
  Thus, evaluating the proximal operator of a separable function reduces to evaluating the proximal operators for each of the separable parts, which can be done independently \cite{parikh_2014}.
\end{proposition}
\begin{proposition}(Proximal operator of norms) \label{normprox}
  If $f = \| \cdot \|$ is a general norm, then $f^*$ is the indicator function on the unit ball for the dual norm $\| \cdot \|_*$, namely 
   \begin{equation*}
    \begin{split}
      \|\cdot \| ^* = \delta_{\|\cdot \|_* \leq 1}
    \end{split}
  \end{equation*}
  where $\|\cdot \|_*$ is given by 
\begin{equation*}
  \begin{split}
    \| v\|_* = \sup_x \{\langle v, x\rangle \mid \|x\|\leq 1\} 
  \end{split}
\end{equation*}
In particular, the \textit{Moreau Decomposition Theorem} gives that 
\begin{equation*}
  \begin{split}
    \prox_{\|\cdot \|}(v) = \proj_{\{x \;: \; \|x \|_* \leq 1\}} (v) - v
  \end{split}
\end{equation*}
\end{proposition}
\section{Algorithms}
The algorithms we present are used to solve the non-blind image deblurring problem, which is a convex optimization problem of the form 
\begin{equation}
  \begin{split}
    \min_{x,y} f(x) + g(y)\\ 
    \text{subject to }Ax= y,
  \end{split}
\end{equation}
where $f, g \in \Gamma$ have inexpensive prox-operators. We are given a vectorized, blurred image $b$, along with the kernel $K$ used to blur it using the model 
\begin{equation}
  Kx + \eta = b,
\end{equation}
where $x$ is the original image, and $\eta$ the added noise. Our goal is to recover the original image $x$. We formulate the problem in two ways, namely, one which uses the $L^1$ norm, and the other the $L^2$ norm. First,
\begin{equation}\label{l1}
    \min_x \|Kx - b\|_1 + \delta_S(x) + \gamma \|Dx \|_{\iso}, 
\end{equation}
and second,
\begin{equation}\label{l2}
    \min_x \|Kx - b\|_2^2 + \delta_S(x) + \gamma \|Dx \|_{\iso}, 
\end{equation}
where $S = \{x \mid 0 \leq x \leq 1\}$, so that $\delta_S$ aims to restrict the range of allowed pixels. The $\iso$ norm is defined as 
$$\|(x,y) \|_{\iso} = \sum^{n}_{k=1} (x_k^2 + y^2_k )^{\frac{1}{2}} $$

\subsection{Primal Douglas-Rachford splitting (Spingarn's method)}%
  \label{sub:Primal Douglas-Rachford splitting (Spingarn's method)}
  This first algorithm, which we will refer to by PDRS, was invented in the 1950s as a method to solve the heat equation \cite{douglas_rachford_1956}. Per (\ref{l1}), let 
  \begin{equation}\label{gf}
    \begin{split}
      f(x) =& \delta_S(x)\\ 
      g(x) =& \|Kx - b\|_1 + \gamma \|Dx\|_{\iso}
    \end{split}
  \end{equation}
  and consider the minimization problem 
  \begin{equation}
    \begin{split}
      \min_{x,y} f(x) + g(y) +\delta_0 \left( \begin{bmatrix}
        A & -I
      \end{bmatrix} \begin{bmatrix}
        x\\y
      \end{bmatrix} \right)
    \end{split}
  \end{equation}
  with the optimality condition 
  \begin{equation}
    \begin{split}
      0 \in \partial f(x) + \partial g(y) + \partial \delta_0 \left( \begin{bmatrix}
        A & -I
      \end{bmatrix} \begin{bmatrix}
        x\\y
      \end{bmatrix} \right) =: \mathcal{F}(x,y) 
    \end{split}
  \end{equation}
  As stated in the literature \cite{ConnorVander}, $\mathcal{F}$ is the subdifferential of closed, convex functions, hence is a maximal monotone operator. Thus, by (\ref{maxmon}), the zeros of $\mathcal{F}$ are the fixed points of the resolvent of $\mathcal{F}$. Therefore, if we can find the fixed points of $\mathcal{F}$, we'll have found the optimal points. Unfortunately, this particular problem is too computationally expensive, so the strategy is to split $\mathcal{F}$ into two maximal monotone operators $\mathcal{A}$ and $\mathcal{B}$ which have inexpensive proximal operators. Finding the fixed points of $\mathcal{A}+ \mathcal{B}$ will be equivalent to finding that of $\mathcal{F}.$ Define $\mathcal{A}$ and $\mathcal{B}$ as, 
  \begin{equation}
    \begin{split}
      \mathcal{A}(x,y)& = \partial f(x) + \partial g(y) = \partial (\delta_S (x)) +\partial (\|Ky - b\|_1 + \gamma \|Dy\|_{\iso})\\ 
      \mathcal{B}(x,y) &= \partial \delta_0 \left( \begin{bmatrix}
        \begin{bmatrix}
          K \\D
        \end{bmatrix} & -I
      \end{bmatrix} \begin{bmatrix}
        x\\y
      \end{bmatrix} \right)
    \end{split}
  \end{equation}
  We now focus on finding the resolvents of $\mathcal{A}$ and $\mathcal{B}$,  
  \begin{equation}
    \begin{split}
      J_{\mathcal{A}} = (I + \lambda \mathcal{A})^{-1} &= \prox_{\lambda (f+g) }(x,y)\quad [\text{by (\ref{subProx})}] \\
      &= (\prox_{\lambda f}(x) ,\prox_{\lambda g}(y))\quad [\text{by Prop. \ref{proxsplit}}]
    \end{split}
  \end{equation}
  where 
  \begin{equation}
    \begin{split}
      \prox_{\lambda f} (x) =& \prox_{\lambda \delta_S}(x) = \argmin_u \frac{1}{2} \| x-u \|^2_2 + \delta_S(x)= \argmin_{u \in S} \frac{1}{2} \| x-u \|^2_2 = \proj_S(x),
    \end{split}
  \end{equation}
  which is the euclidean projection operator onto $S.$ It can be computed element-wise by
  \begin{equation}
    \begin{split}
      (\proj_S(x))_i = 
      \begin{cases}
        0, &\text{ if } x_i < 0\\
        x_i, &\text{ if } x_i \in S\\ 
        1, & \text{ if } x_i > 1 
      \end{cases} = \max \{0 , \min\{x_i , 1\}\}
          \end{split}
  \end{equation}


  Now, we observe that $g(y)$ described in  (\ref{gf}) can be written in a different form by letting  
  \begin{equation}
    \begin{split}
      g(y_1, y_2, y_3) = \| y_1 -b \|_{1} + \gamma \|(y_2, y_3) \|_{\iso}
    \end{split}
  \end{equation}
  subject to 
  $$y_1 = Ky\quad \text{ and }\quad \begin{bmatrix}
    y_2 \\ y_3 
  \end{bmatrix} = Dy \iff \begin{bmatrix}
    y_1\\ y_2 \\ y_3
  \end{bmatrix} = \begin{bmatrix}
    K\\ D 
  \end{bmatrix} y  $$ 
  Then, by Proposition \ref{proxsplit}, notice that the proximal operator of $g$ may be obtained by splitting its components into 
  \begin{equation*}
    \begin{split}
      \varphi(y_1) = \|y_1 -b \|_1 \quad \text{ and } \quad \psi(y_2, y_3)= \gamma \| (y_2, y_3)\|_{\iso}
    \end{split}
  \end{equation*}
  which gives 
  \begin{equation}
    \begin{split}
      \prox_{\lambda g}(y_1, y_2, y_3) =&(\prox_{\lambda \varphi}(y_1), \prox_{\lambda \psi}(y_2, y_3))
    \end{split}
  \end{equation}
  We know that the proximal operator of the $L^1$ norm is the soft-thresholding operator $T_\lambda$, so $\prox_{\lambda \varphi} (y_1)$ is given element-wise as follows, 
  \begin{equation}
       (\prox_{\lambda \varphi} (y_1))_i = (\prox_{\lambda \|\cdot\|_1}(y_1-b))_i= T_\lambda({(y_1-b)_i}) :=  
      \begin{cases}
        (y_1-b)_i - \lambda, &\text{ if } (y_1-b)_i >\lambda\\ 
        (y_1-b)_i+ \lambda, &\text{ if }(y_1-b)_i <-\lambda  \\ 
        0, &\text{ otherwise. }
      \end{cases}
  \end{equation}
  As a result of Proposition \ref{normprox}, $\prox_{\lambda \psi}(y_2, y_3)$, is given element-wise by 
  \begin{equation*}
    \begin{split}
      (\prox_{\lambda \psi}(y_2, y_3))_i =  \alpha_i ((y_2)_i, (y_3)_i), 
    \end{split}
  \end{equation*}
  where, \marginpar{\color{red}(provide a proof of this)}
  \begin{equation*}
    \begin{split}
      \alpha_i = 
      \begin{cases}
        1 - \frac{\lambda }{\sqrt{(y_2)_i^2 + (y_3)_i^2}},  &\text{ if }\sqrt{(y_2)_i^2 + (y_3)_i^2}> \lambda \\ 
         0, &\text{ otherwise. }
      \end{cases}
    \end{split}
  \end{equation*}
  As for the resolvent of $\mathcal{B}$, it corresponds to the projection $(x, y)$ of a vector $(\bar x, \bar y)$ onto the subspace $\{(x, y) \mid
\begin{bmatrix}
  K &D
        \end{bmatrix}^Tx = y\}$, expressed as
 \begin{equation*}
  \begin{split}
    J_{\mathcal{B}} = (I + \lambda \mathcal{B})^{-1} =\left(x=\left(I+ \begin{bmatrix}
      K &D
        \end{bmatrix} \begin{bmatrix}
          K \\D
        \end{bmatrix}\right)^{-1}\left(\bar x +  \begin{bmatrix}
          K \\D
        \end{bmatrix}\bar y\right), y= \begin{bmatrix}
          K \\D
        \end{bmatrix}x \right)
  \end{split}
 \end{equation*}
