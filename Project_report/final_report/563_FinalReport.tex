\begin{abstract}
  The frustrating problem of \textit{blurry} images has existed since the invention of cameras. Multiple types of blurs occur for different reasons, such as \textit{camera shake blur}, \textit{motion blur}, or \textit{out of focus blur}. These images are a nuisance, and pose challenges in many applications, such as image content recognition, surveillance, astronomy and many more \cite{wang2014recent}. Hence, reducing blur, or \textit{deblurring} is crucial in solving the ubiquitous problem of blurred images in a society inundated with visual media. In this continually evolving research area, many methods have been proposed, broadly categorized into those addressing the \textit{non-blind} and \textit{blind} deblurring problems. This article will primarily focus on the former, where the blur kernel is known. We will discuss four algorithms designed to tackle this challenge, highlighting their strengths and weaknesses while providing illustrative examples.
\end{abstract}

\section{Introduction}
\subsection{Notation}%
  \label{sub:Notation}
  Throughout this paper, we will denote the \textit{proximal operator} of $f$ with parameter $\lambda$ by 
  \begin{equation}
    \begin{split}
      \prox_{\lambda f}(v) := \argmin_x \left\{f(x) + \frac{1}{2\lambda} \| x - v\|_2^2 \right\}
    \end{split}
  \end{equation}
  The \textit{indicator function} for a convex set $C$ will be defined as 
  \begin{equation}
    \begin{split}
      \delta_C (x) := 
      \begin{cases}
        0, &\text{ if } x \in C\\
        +\infty, &\text{ if } x \notin C
      \end{cases}
    \end{split}
  \end{equation}
  We also denote the \textit{Euclidean Projection} on a set $C$ by 
  \begin{equation}
    \begin{split}
      \proj_C(v):= \argmin_{x\in C} \| x -v \|_2
    \end{split}
  \end{equation}
  
\subsection{Preleminaries}%
  \label{sub:Preleminaries}
  We first note an important relation between the proximal and subdifferential operators.\\
  \begin{proposition}
    Given a function $f\in\Gamma$, we have 
    \begin{equation}\label{subProx}
      \prox_{\lambda f} =(I+ \lambda \partial f)^{-1},
    \end{equation}  
    where $(I + \lambda \partial f)^{-1}$, for $\lambda > 0$, is called the \textbf{resolvent} of the operator $\partial f$. In particular, the proximal operator is the resolvent of the subdifferential operator \cite{parikh_2014}.\\
  \end{proposition}
  \begin{definition}[Maximal Monotone operator]
    A monotone operator is \textbf{maximal} if its graph is not properly contained in the graph of any other monotone operator. If A is monotone (resp. maximal monotone) then so are $A^{-1}$ and $\lambda A$ if $\lambda > 0$.
  \end{definition}
  The next proposition, given in \cite{ConnorVander}, will be useful in the construction of the optimizing algorithms.\\ 
  \begin{proposition}     
If $F$ is maximal monotone, then $(I + \lambda F)^{-1}(\hat x)$ exists and is unique for all $\hat x \in R^n$. The value $x = (I + \lambda F)^{-1}(\hat x)$ of the resolvent is the unique solution of the monotone inclusion
    \begin{equation}\label{maxmon}
      \hat x \in x + \lambda {F}(x) 
   \end{equation}
  \end{proposition}  
\begin{proposition}[Proximal operator splitting] \label{proxsplit}
  If $f$ is separable across two variables, so $f(x, y) = \varphi(x) + \psi(y)$, then
  \begin{equation}
    \prox_f (x,y) = (\prox_\varphi(x), \prox_\psi(y))
  \end{equation}
  Thus, evaluating the proximal operator of a separable function reduces to evaluating the proximal operators for each of the separable parts, which can be done independently \cite{parikh_2014}.
\end{proposition}

\begin{proposition}(Proximal operator of norms) \label{normprox}
  If $f = \| \cdot \|$ is a general norm, then $f^*$ is the indicator function on the unit ball for the dual norm $\| \cdot \|_*$, namely 
   \begin{equation*}
    \begin{split}
      \|\cdot \| ^* = \delta_{\|\cdot \|_* \leq 1}
    \end{split}
  \end{equation*}
  where $\|\cdot \|_*$ is given by 
\begin{equation*}
  \begin{split}
    \| v\|_* = \sup_x \{\langle v, x\rangle \mid \|x\|\leq 1\} 
  \end{split}
\end{equation*}
In particular, the \textit{Moreau Decomposition Theorem} gives that 
\begin{equation*}
  \begin{split}
    \prox_{\|\cdot \|}(v) = v - \proj_{\{x \;: \; \|x \|_* \leq 1\}} (v) 
  \end{split}
\end{equation*}
\end{proposition}

\subsection{Derivations of the proximal maps}
We provide a quick derivation of the proximal operators of the L1 norm, the L2 norm, the iso norm, and the box prox. By the Moreau decomposition, we naturally obtain the proximal operators of the Fenchel conjugates. Let $b \in \R^d$ be fixed.

\medskip

\begin{itemize}[leftmargin=*]
    \item Let $ g: y \mapsto t \| y - b \|_1$. We have
    \begin{align*}
        \prox_{g}(y) &:= \argmin_u \{ \: t \| u - b \|_1 + \frac{1}{2} \| y - u \|^2_2 \: \} =
        \begin{cases}
            b & y - b \in [-t, t] \\
            y - \text{sign}(y-b) \cdot t &\text{otherwise}
        \end{cases}
    \end{align*}
    We consider first-order optimality conditions, with the following cases:
    \begin{itemize}
        \item If $u - b > 0$, then 
        \begin{align*}
            t - y + u = 0 \quad \Leftrightarrow \quad u = y - t
        \end{align*}
        \item If $u - b < 0$, then 
        \begin{align*}
            -t - y + u = 0 \quad \Leftrightarrow \quad u = y + t
        \end{align*}
        \item If $u - b = 0$, then $u = b$ under which by optimality conditions we have
        \begin{align*}
            0 \in t \cdot [-1, 1] - y + b \quad \Leftrightarrow \quad y - b \in [-t, t]
        \end{align*}
    \end{itemize}

    \smallskip

    \item Now consider $g: y \mapsto t \| y - b \|_2^2$. This $g$ is continuously differentiable so we can take its gradient directly and set it to zero. That is, the steps are:
    \begin{align*}
        \frac{\partial}{\partial u} \bigg( t \| u - b \|_2^2 + \frac{1}{2}\|y - u \|_2^2 \bigg) = 2 t (u - b) - y + u = u(2t + 1) - y - 2tb
    \end{align*}
    Therefore
    $$
    \prox_{g}(y) = \frac{y + 2tb}{2 t + 1}
    $$
    \smallskip

    \item For $g: (y_1, y_2) \mapsto t\gamma \| (y_1, y_2) \|_{\text{iso}}$ where
    $$
    \|(x,y) \|_{\iso} = \sum^{n}_{k=1} (x_k^2 + y^2_k )^{1/2} 
    $$
    To compute the proximal operator of $g$, we use Moreau decomposition (7.5.8). The Fenchel conjugate of $g$ is $g^*(y_1, y_2) = t \gamma \max_{1 \leq k \leq n} (y_{1k}^2 + y_{2k}^2)^{1/2}$ where $y_1, y_2 \in \mathbb{R}^n$ (simplified after elementary conjugacy operations). So we have
    \begin{align*}
        \prox_{g}(y_1, y_2) = (y_1, y_2) - \prox_{g^*}(y_1, y_2)
    \end{align*}
    where
    \begin{align*}
        \prox_{g^*}(y_1, y_2) = \argmin_{(u, v)} \bigg{\{} t \gamma \max_{1 \leq k \leq n} (u_k^2 + v_k^2)^{1/2}  + \frac{1}{2} \| (y_1, y_2) - (u, v) \|_2^2 \bigg{\}}
    \end{align*}
    \begin{itemize}
        Remark that $(u_k^2 + v_k^2)^{1/2} = \| (u_k, v_k) \|^2_2$ for each $k$. The Fenchel conjugate of the L2 norm is itself (HW3). We also have that for any norm $\| \cdot \|$, the Fenchel conjugate $\| \cdot \|^*$ is the indicator of the unit ball in the dual norm $\delta_{\| \cdot \|_* \leq 1}$. The proximal mapping of the indicator of the unit ball is the projection onto the unit ball for points not in the unit ball. Hence
    \end{itemize}
    $$
    \prox_{g^*}(y_1, y_2) = t \gamma \frac{(y_1, y_2)}{\| (y_1, y_2) \|_2}
    $$
    as long as $(y_{1k}^2 + y_{2k}^2)^{1/2} > 0$. If $(y_{1k}^2 + y_{2k}^2)^{1/2} = 0$, then $y_{1k} = y_{2k} = 0$ so $(u, v) = (0, 0)$. So altogether, for $k= 1,...,n$
    $$
    \prox_{g}(y_{1k}, y_{2k}) = 
    \begin{cases}
        \bigg( 1 - \frac{t \gamma}{(y_{1k}^2 + y_{2k}^2)^{1/2}} \bigg) (y_{1k}, y_{2k}) &(y_{1k}^2 + y_{2k}^2)^{1/2} > 0 \\
        0 & \text{otherwise}
    \end{cases}
    $$

    \smallskip
    
    \item For the box prox, $f : x \mapsto t \delta_S(x)$ where $S := \{x : x \in [0, 1]\}$, the proximal operator is the projection. That is, 
    \begin{align*}
        \prox_{f}(x) &= \argmin_u \{ t \delta_S(u) + \frac{1}{2}\|x - u\|^2_2 \} \\
        &= t \cdot \argmin_u \{ \delta_S(u) + \frac{1}{2t}\|x - u\|^2_2 \} \\
        &= \argmin_{u \in S} \frac{1}{2}\|x - u\|^2_2
    \end{align*}
\end{itemize}

\section{Algorithms}
\label{algos}
The algorithms we present are used to solve the non-blind image deblurring problem, which is a convex optimization problem of the form 
\begin{equation}
  \begin{split}
    \min_{x,y} f(x) + g(y)\\ 
    \text{subject to }Ax= y,
  \end{split}
\end{equation}
where $f, g \in \Gamma$ have inexpensive prox-operators. We are given a vectorized, blurred image $b$, along with the kernel $K$ used to blur it using the model 
\begin{equation}
  Kx + \eta = b,
\end{equation}
where $x$ is the original image, and $\eta$ the added noise. Our goal is to recover the original image $x$ from the blurred and noisy image $b$. We formulate the problem in two ways, namely, one which uses the $L^1$ norm, and the other the $L^2$ norm. First,
\begin{equation}\label{l1}
    \min_x \|Kx - b\|_1 + \delta_S(x) + \gamma \|Dx \|_{\iso}, 
\end{equation}
and second,
\begin{equation}\label{l2}
    \min_x \|Kx - b\|_2^2 + \delta_S(x) + \gamma \|Dx \|_{\iso}, 
\end{equation}
where $S = \{x \mid 0 \leq x \leq 1\}$, so that $\delta_S$ aims to restrict the range of allowed pixels. 

\medskip

The algorithms are based on a splitting scheme to facilitate computation of the resolvents of the maximal monotone operators. As mentioned in a bit more detail in the project prompt, we have from the fixed point iteration scheme
$$
x^k = (I + t \mathcal{F})^{-1}(x^{k-1}) \qquad k = 1, 2, ...
$$
where $\mathcal{F}$ is a maximal monotone operator. Splitting algorithms decompose the problem into
\begin{align*}
    &x^k = (I + t \mathcal{A})^{-1}(z^{k-1}) \\
    &y^k = (I + t \mathcal{B})^{-1}(2x^{k} - z^{k-1}) \\
    &z^k = z^{k-1} \rho(y^k - x^k)
\end{align*}
where $t > 0$ is the step size and $\rho \in (0, 2)$ is a relaxation parameter. Splitting algorithms are used en lieu of a classic fixed point iteration due to expensive computation costs.

\subsection{Primal Douglas-Rachford splitting (Spingarn's method)}%
  \label{sub:PDRS}
  This first algorithm, which we will refer to by PDRS, was invented in the 1950s as a method to solve the heat equation \cite{douglas_rachford_1956}. Per (\ref{l1}), let 
  \begin{equation}\label{gf}
    \begin{split}
      f(x) =& \delta_S(x)\\ 
      g(x) =& \|Kx - b\|_1 + \gamma \|Dx\|_{\iso}
    \end{split}
  \end{equation}
  and consider the minimization problem 
  \begin{equation}
    \begin{split}
      \min_{x,y} f(x) + g(y) +\delta_0 \left( \begin{bmatrix}
        A & -I
      \end{bmatrix} \begin{bmatrix}
        x\\y
      \end{bmatrix} \right)
    \end{split}
  \end{equation}
  with the optimality condition 
  \begin{equation*}
    \begin{split}
      0 \in \partial f(x) + \partial g(y) + \partial \delta_0 \left( \begin{bmatrix}
        A & -I
      \end{bmatrix} \begin{bmatrix}
        x\\y
      \end{bmatrix} \right) =: \mathcal{F}(x,y) 
    \end{split}
  \end{equation*}
  As stated in the literature \cite{ConnorVander}, $\mathcal{F}$ is the subdifferential of closed, convex functions, hence is a maximal monotone operator. Thus, by (\ref{maxmon}), the zeros of $\mathcal{F}$ are the fixed points of the resolvent of $\mathcal{F}$. Therefore, if we can find the fixed points of $\mathcal{F}$, we'll have found the optimal points. Unfortunately, this particular problem is too computationally expensive, so the strategy is to split $\mathcal{F}$ into two maximal monotone operators $\mathcal{A}$ and $\mathcal{B}$ which have inexpensive proximal operators. Finding the fixed points of $\mathcal{A}+ \mathcal{B}$ will be equivalent to finding that of $\mathcal{F}.$ Define $\mathcal{A}$ and $\mathcal{B}$ as, 
  \begin{equation}
    \begin{split}
      \mathcal{A}(x,y)& = \partial f(x) + \partial g(y) = \partial (\delta_S (x)) +\partial (\|Ky - b\|_1 + \gamma \|Dy\|_{\iso})\\ 
      \mathcal{B}(x,y) &= \partial \delta_0 \left( \begin{bmatrix}
        \begin{bmatrix}
          K \\D
        \end{bmatrix} & -I
      \end{bmatrix} \begin{bmatrix}
        x\\y
      \end{bmatrix} \right)
    \end{split}
  \end{equation}
  We now focus on finding the resolvents of $\mathcal{A}$ and $\mathcal{B}$,  
  \begin{equation}
    \begin{split}
      J_{\mathcal{A}} = (I + \lambda \mathcal{A})^{-1} &= \prox_{\lambda (f+g) }(x,y)\quad [\text{by (\ref{subProx})}] \\
      &= (\prox_{\lambda f}(x) ,\prox_{\lambda g}(y))\quad [\text{by Prop. \ref{proxsplit}}]
    \end{split}
  \end{equation}
  where 
  \begin{equation}
    \begin{split}
      \prox_{\lambda f} (x) =& \prox_{\lambda \delta_S}(x) = \argmin_u \frac{1}{2\lambda} \| x-u \|^2_2 + \delta_S(x)= \argmin_{u \in S} \frac{1}{2\lambda} \| x-u \|^2_2 =  \proj_S(x),
    \end{split}
  \end{equation}
  which is the euclidean projection operator onto $S.$ It can be computed element-wise by
  \begin{equation}
    \begin{split}
      (\proj_S(x))_i = 
      \begin{cases}
        0, &\text{ if } x_i < 0\\
        x_i, &\text{ if } x_i \in S\\ 
        1, & \text{ if } x_i > 1 
      \end{cases} = \max \{0 , \min\{x_i , 1\}\}
          \end{split}
  \end{equation}


  Now, we observe that $g(y)$ described in  (\ref{gf}) can be written in a different form by letting  
  \begin{equation}
    \begin{split}
      g(y_1, y_2, y_3) = \| y_1 -b \|_{1} + \gamma \|(y_2, y_3) \|_{\iso}
    \end{split}
  \end{equation}
  subject to 
  $$y_1 = Ky\quad \text{ and }\quad \begin{bmatrix}
    y_2 \\ y_3 
  \end{bmatrix} = Dy \iff \begin{bmatrix}
    y_1\\ y_2 \\ y_3
  \end{bmatrix} = \begin{bmatrix}
    K\\ D 
  \end{bmatrix} y  $$ 
  Then, by Proposition \ref{proxsplit}, notice that the proximal operator of $g$ may be obtained by splitting its components into 
  \begin{equation*}
    \begin{split}
      \varphi(y_1) = \|y_1 -b \|_1 \quad \text{ and } \quad \psi(y_2, y_3)= \gamma \| (y_2, y_3)\|_{\iso}
    \end{split}
  \end{equation*}
  which gives 
  \begin{equation}
    \begin{split}
      \prox_{\lambda g}(y_1, y_2, y_3) =&(\prox_{\lambda \varphi}(y_1), \prox_{\lambda \psi}(y_2, y_3))
    \end{split}
  \end{equation}
  We know that the proximal operator of the $L^1$ norm is the soft-thresholding operator $T_\lambda$, so $\prox_{\lambda \varphi} (y_1)$ is given element-wise as follows, 
  \begin{equation}
       (\prox_{\lambda \varphi} (y_1))_i = (\prox_{\lambda \|\cdot\|_1}(y_1-b))_i= T_\lambda({(y_1-b)_i}) :=  
      \begin{cases}
        (y_1-b)_i - \lambda, &\text{ if } (y_1-b)_i >\lambda\\ 
        (y_1-b)_i+ \lambda, &\text{ if }(y_1-b)_i <-\lambda  \\ 
        0, &\text{ otherwise. }
      \end{cases}
  \end{equation}
  As a result of Proposition \ref{normprox}, $\prox_{\lambda \psi}(y_2, y_3)$, is given element-wise by 
  \begin{equation*}
    \begin{split}
      (\prox_{\lambda \psi}(y_2, y_3))_i = \prox_{\gamma \lambda \| \cdot \|_{\iso}} (y_2, y_3)_{i}= \alpha_i ((y_2)_i, (y_3)_i), 
    \end{split}
  \end{equation*}
  where,
  \begin{equation*}
    \begin{split}
      \alpha_i = 
      \begin{cases}
        1 - \frac{\gamma \lambda }{\sqrt{(y_2)_i^2 + (y_3)_i^2}},  &\text{ if }\sqrt{(y_2)_i^2 + (y_3)_i^2}> \gamma \lambda \\ 
         0, &\text{ otherwise. }
      \end{cases}
    \end{split}
  \end{equation*}
  As for the resolvent of $\mathcal{B}$, it corresponds to the projection $(x, y)$ of a vector $(\bar x, \bar y)$ onto the subspace $\{(x, y) \mid
\begin{bmatrix}
  K &D
        \end{bmatrix}^Tx = y\}$, expressed as
 \begin{equation*}
  \begin{split}
    J_{\mathcal{B}} = (I + \lambda \mathcal{B})^{-1} =\left(x=\left(I+ \begin{bmatrix}
      K &D
        \end{bmatrix} \begin{bmatrix}
          K \\D
        \end{bmatrix}\right)^{-1}\left(\bar x +  \begin{bmatrix}
          K \\D
        \end{bmatrix}\bar y\right), y= \begin{bmatrix}
          K \\D
        \end{bmatrix}x \right)
  \end{split}
 \end{equation*}
 The algorithm is included in Appendix \ref{sub:pseudo}.

\subsection{Primal-Dual Douglas Rachford Splitting}\label{sub:Primal-Dual Douglas Rachford splitting}
This second method uses both optimality conditions of the primal problem
and the dual problem for the "split". The method takes advantage of the inexpensive computation of the prox-operators of $f$ and $g$, and the fact that our matrix $A$ makes it easy to compute linear equations.

In the algorithm, we consider the dual problem, 
\begin{equation}
  \text{maximize} \ -f^*(-A^T z) - g^*(z)
\end{equation}
with the optimality conditions for both the primal and the dual problem 

\[
0 \in 
\begin{bmatrix}
  0 & 0 & A^T & I \\
  0 & 0 & -I & 0 \\
  -A & I & 0 & 0 \\
  -I & 0 & 0 & 0 \\
\end{bmatrix}
\begin{bmatrix}
  x \\
  y \\ 
  z \\
  w \\
\end{bmatrix} 
+ 
\begin{bmatrix}
  0 \\
  \partial g(y) \\
  0 \\
  \partial f^*(w) \\
\end{bmatrix}
\]

In the above, $w$ is the dual variable and $z$ is the dual multiplier for the constraint $y = Ax$, thus the dual problem is subject to
\begin{equation*}
  A^T z + w = 0
\end{equation*}

We reduce this to a smaller system by using the relationship $\partial g^* = {(\partial g)}^{-1}$, which follows from the fact that $g$ is closed and convex, and by the properties of the subgradient and conjugate. We can eliminate $y$ by using the above equality that $\partial g^* = {(\partial g)}^{-1}$, which would leave us with
\[
0 \in 
\begin{bmatrix}
  0 & A^T & I \\
  -A & 0 & 0 \\
  -I & 0 & 0 \\
\end{bmatrix}
\begin{bmatrix}
  x \\
  z \\
  w \\
\end{bmatrix}  
+
\begin{bmatrix}
  0 \\
  \partial g^*(z) \\
  \partial f^*(w) \\
\end{bmatrix}
\]

We can further reduce this by eliminating $w$, using a similar equality that $\partial f = {(\partial f^*)}^{-1}$, we get

\begin{equation}
  0 \in 
  \begin{bmatrix}
    0 & A^T \\
    -A & 0 \\
  \end{bmatrix}
  \begin{bmatrix}
    x \\
    z \\
  \end{bmatrix}  
  + 
  \begin{bmatrix}
    \partial f(x) \\
    \partial g^*(z) \\
  \end{bmatrix}
\end{equation}


We can now split the monotone operator we have above to a sum of two monotone operators given as 
\begin{equation}
  \mathcal{A}(x, z) = 
  \begin{bmatrix}
    \partial f(x) \\
    \partial g^*(x) \\
  \end{bmatrix}
\quad
\mathcal{B}(x, z) = 
\begin{bmatrix}
  0 & A^T \\
  -A & 0 \\
\end{bmatrix}
\begin{bmatrix}
  x \\
  z \\
\end{bmatrix}
\end{equation}

Similar to the previous algorithm, we now find the resolvents of $\mathcal{A}$ and $\mathcal{B}$. 

As seen before with the resolvent of $\mathcal{A}$, the value $(x, y) = {(I + At)}^{-1}(\hat{x}, \hat{z})$ is given by 
\begin{equation*}
  x = \prox_{t f}(\hat{x}) \quad z = \prox_{tg^*}(\hat{z})
\end{equation*}

For $z$, we can apply moreau decomposition to get an alternate expression dealing with the conjugate $g^*$, thus we have, 
\begin{equation*}
  z = \hat{z} - t \prox_{t^{-1}g}(\nicefrac{\hat{z}}{t})
\end{equation*}
Simplifying the above, we are left with, 
\begin{equation*}
  z = \hat{z} - \prox_{gt}(\hat{z})
\end{equation*}

This will be a similar computation to the prox operator in algorithm 1, but we have right scalar multiplication in this case. 

The resolvent for $\mathcal{B}$ is a linear mapping, so considering resolvents for skew-symmetric linear operators 
\begin{align*}
  {(I + t\mathcal{B})}^{-1} = 
  {\begin{bmatrix}
    I & tA^T \\
    -tA & I \\
  \end{bmatrix}}^{-1} =
  \frac{1}{I - ((tA^T)(-tA))}
  \begin{bmatrix}
    I & -tA^T \\
    tA & I \\
  \end{bmatrix} &=
  \frac{1}{I + t^2A^T A}
  \begin{bmatrix}
    I & -tA^T \\
    tA & I \\
  \end{bmatrix} \\
  &=
  \begin{bmatrix}
    0 & 0 \\
    0 & I \\
  \end{bmatrix} + 
  \frac{1}{I + t^2A^T A}
  \begin{bmatrix}
    I \\
    tA \\
  \end{bmatrix}
  \begin{bmatrix}
    I & -tA^T \\
  \end{bmatrix} \\
  &=
  \begin{bmatrix}
    0 & 0 \\
    0 & I \\
  \end{bmatrix} +
  \begin{bmatrix}
    I \\
    tA \\
  \end{bmatrix}
  {(I + t^2A^T A)}^{-1}
  {\begin{bmatrix}
    I \\
    -tA \\
  \end{bmatrix}}^T \\
\end{align*}

Letting $(w, v)$ be the value of the resolvent of $\mathcal{B}$, 
\begin{align*}
  \begin{bmatrix}
    w \\
    v \\
  \end{bmatrix} 
  =
  {\begin{bmatrix}
    I & tA^T \\
    -tA & I \\
  \end{bmatrix}}^{-1}
  \begin{bmatrix}
    \hat{x} \\
    \hat{z} \\
  \end{bmatrix} 
  &= \left(
  \begin{bmatrix}
    0 & 0 \\
    0 & I \\
  \end{bmatrix} +
  \begin{bmatrix}
    I \\
    tA \\
  \end{bmatrix}
  {(I + t^2A^T A)}^{-1}
  {\begin{bmatrix}
    I \\
    -tA \\
  \end{bmatrix}}^T \right)
  \begin{bmatrix}
    \hat{x} \\
    \hat{z} \\
  \end{bmatrix}\\
  &= \begin{bmatrix}
    0 & 0 \\
    0 & I \\
  \end{bmatrix}
  \begin{bmatrix}
    \hat{x} \\
    \hat{z} \\
  \end{bmatrix}
  + 
  \begin{bmatrix}
    I \\
    tA \\
  \end{bmatrix}
  {(I + t^2A^T A)}^{-1}
  {\begin{bmatrix}
    I \\
    -tA \\
  \end{bmatrix}}^T
  \begin{bmatrix}
    \hat{x} \\
    \hat{z} \\
  \end{bmatrix} \\
  &= 
  \begin{bmatrix}
    0 \\
    \hat{z} \\
  \end{bmatrix}
  + 
  \begin{bmatrix}
    I \\
    tA \\
  \end{bmatrix}
  {(I + t^2A^T A)}^{-1}
  \begin{bmatrix}
    I & -tA^T \\
  \end{bmatrix}
  \begin{bmatrix}
    \hat{x} \\
    \hat{z} \\
  \end{bmatrix} \\
  &= 
  \begin{bmatrix}
    0 \\
    \hat{z} \\
  \end{bmatrix}
  + 
  \begin{bmatrix}
    I \\
    tA \\
  \end{bmatrix}
  {(I + t^2A^T A)}^{-1}
  (\hat{x} - tA^T \hat{z}) \\
  &= 
  \begin{bmatrix}
    0 \\
    \hat{z} \\
  \end{bmatrix}
  + 
  \begin{bmatrix}
    I \\
    tA \\
  \end{bmatrix}
  \frac{\hat{x} - tA^T \hat{z}}{(I + t^2A^T A)} \\
  &=
  \begin{bmatrix}
    0 \\
    \hat{z} \\
  \end{bmatrix}
  +
  \begin{bmatrix}
    \frac{\hat{x} - tA^T \hat{z}}{(I + t^2A^T A)} \\
    tA \left(\frac{\hat{x} - tA^T \hat{z}}{(I + t^2A^T A)}\right) \\
  \end{bmatrix} = 
  \begin{bmatrix}
    \frac{\hat{x} - tA^T \hat{z}}{(I + t^2A^T A)} \\
    \hat{z} + tA \left(\frac{\hat{x} - tA^T \hat{z}}{(I + t^2A^T A)}\right) \\
  \end{bmatrix} \\
\end{align*}
Thus we have the solution for both $w, \ v$, 
\begin{align}
  w = \frac{\hat{x} - tA^T \hat{z}}{(I + t^2A^T A)} = \frac{\hat{x}}{(I + t^2A^T A)} - \frac{tA^T \hat{z}}{(I + t^2A^T A)}
\end{align}
\begin{align}
  v &= \hat{z} + tA \left(\frac{\hat{x} - tA^T \hat{z}}{(I + t^2A^T A)}\right) = \hat{z} + \frac{\hat{tAx}}{(I + t^2A^T A)} - \frac{t^2 A A^T \hat{z}}{(I + t^2A^T A)}
\end{align}

Notice in the algorithm, we will let $\hat{x} = 2x^k - p^{k-1}$ and $\hat{z} = 2z^k - p^{k-1}$. The Primal-Dual Douglas Rachford splitting algorithm is included in Appendix \ref{sub:pseudo}.

This algorithm resembles the first algorithm in section \ref{sub:PDRS}. They are both splitting algorithm that use the prox-operators of $f$ and $g$. The algorithms differ in the fact that the Primal-Dual Douglas Rachford splitting algorithm uses both the primal and dual optimality conditions, rather than uniquely using the primal as done in section \ref{sub:PDRS}. This means that we need to use Moreau decomposition for the congugate of the prox-operator of $g$. The $t$ and $\rho$ inputs play a very similar role in the algorithm as it does in the algorithms in section \ref{sub:PDRS} and \ref{sub: ADMM}. 

\subsection{ADMM (Dual Douglas-Rachford)}  \label{sub: ADMM}
The dual problem from above
\begin{equation}
  \text{maximize}_y \ -f^*(-A^T y) - g^*(y)
\end{equation}
has the equivalent form
$$
\text{minimize}_z \ -f^*(-A^T z) - g^*(z) := h(z) + k(z)
$$
and it is said that applying the Douglas-Rachford algorithm to this dual problem results in the alternating direction method of multipliers (ADMM) algorithm. That is, ADMM uses the Douglas-Rachford iteration on the dual problem. Optimality is guaranteed when $0 \in \partial h(z) + \partial k(z)$ in which we have two maximal monotone operators with resolvents $\prox_{th}(z)$ and $\prox_{tk}(z)$. 

\medskip

ADMM, like the previous algorithms, is a splitting algorithm. Unlike the previous two algorithms, it has different optimality conditions, that is it uses solely optimality conditions of the dual problem. Apart from this, it relies, as the other algorithms do, on the Douglas-Rachford iteration, with a different update ordering as mentioned in the project prompt. The proximal operators are fundamentally the same despite different inputs and despite using the inverse of the step size $t^{-1}$. This is can be interpreted by the reliance of the Lagrangian on functions $f$ and $g$. Another difference is the use of the step size $t$ to update $w^k$ and $z^k$, whereas previously somewhat analogous updates used the relaxation parameter $\rho$. It turns out the $\rho$ is still used, but hidden within the update steps.

\medskip

A form of the Douglas-Rachford iteration on the dual can be derived to be the following, as given in the project prompt:
\begin{align*}
\hat{x}^k &= \underset{x}{\arg \min} \left( f(x) + (z^{k-1} - w^{k-1})^T Ax + \frac{t}{2} \|Ax \|^2 \right) \\
v^k &= z^{k-1} - w^{k-1} + tA \hat{x}^k \\
\hat{y}^k &= \underset{y}{\arg \min} \left( g(y) -(v^k + w^{k-1})^T y + \frac{t}{2} \|y \|^2 \right) \\
z^k &= v^k + w^{k-1} - t \hat{y}^k\\
w^k &= w^{k-1}+ \rho v^k + (1-\rho)z^{k-1} - z^k.
\end{align*}

In particular, we are interested in the general problem
\begin{align*}
    \begin{split}
        \min_{u,y}& \: f(u) + g(y)\\ 
        \text{subject to }&
        \begin{bmatrix}
            I \\
            A
        \end{bmatrix} x + 
        \begin{bmatrix}
            -I & 0 \\
            0 & -I
        \end{bmatrix}
        \begin{bmatrix}
            u \\
            y
        \end{bmatrix} = 0
    \end{split}
\end{align*}
for which we have an augmented Lagrangian given as
$$
L(x,u, y, w, z) = f(u) + g(y) + w^T (x-u) + z^T (Ax-y) + \frac{t}{2} \left( \| x-u\|^2 + \|Ax-y\|^2 \right)
$$
which makes no difference in the above formulation as we alternate minimization over $x$ and minimization over $(u, y)$, so that they are constants in alternation. The pseudo code can be seen in Appendix \ref{sub:pseudo}.

\subsection{Chambolle-Pock}%
  \label{sub:Chambolle-Pock}

The Chambolle-Pock method considers the generic optimization problem 
\begin{equation*}
  \begin{split}
    \min_{x,y} f(x) + g(y)\\ 
    \text{subject to }Ax= y,
  \end{split}
\end{equation*}
by finding the solution of
\begin{equation}
 \min_{x}\max_{y} \, \langle Ax, y \rangle + f(x) - g^{*}(y). \label{eq: CP_minmax}
\end{equation}
It is mentioned that in our case, the solution turns out to be the Lagrangian of our generic optimization problem. 

\medskip

The Chambolle-Pock algorithm stands out as the odd one out among the four algorithms. It does not use a relaxation parameter $\rho$, but two step sizes $t$ and $s$ which depend on the size of the operators in the problem, as mentioned in the project prompt. Furthermore, this algorithm does not require solving a system of linear equations involving $I +A^{T}A$. We only need vector-matrix multiplications of $A$ and $A^{T}$, hence giving a time complexity that is not necessarily the same as the previous algorithms.

\medskip

Chambolle-Pock is not based on an "optimality condition" as the previous three, but properly chosen step sizes $s$ and $t$ that will guarentee convergence. In particular, as described in the project prompt, by choosing the step sizes $s$ and $t$ such that $st(\|A\|_{2}^{2}) < 1$, we can ensure the following convergence:

\begin{equation}
 \frac{\| y^{k} - \hat{y}\|_{2}^{2}}{2s}  + \frac{\| x^{k} - \hat{x} \|_{2}^{2}}{2t} \leq C \left(\frac{\| y^0 - \hat{y}\|_{2}^{2}}{2s}  + \frac{\|x^0 - \hat{x}\|_{2}^{2}}{2t} \right)
\end{equation}

In summary, because the sequence of iterates $\{ (x^k, y^k) \}_k$ is bounded, by extracting subsequences, convergence to a fixed point is guaranteed.

\medskip

Despite these differences, it is said that the Chambolle-Pock algorithm is at its core, a form of preconditioned ADMM. Unpacking this formulation, and with the help of the equations 
\[ y^{k}  = (I + s\partial g^*)^{-1}(y^{k-1} + sAz^{k-1}) \qquad z^k = 2x^k - x^{k-1} \]
due to the addition of 
\[ \frac{1}{2}\left\langle \left(\frac{1}{s} - t AA^T\right)(y - y^{k-1}), y - y^{k-1} \right\rangle \]
to the augmented Lagrangian at the $k$-th iteration, we see that the updates of $y^k$ and $z^k$ are exactly two of the steps of the algorithm described above. The algorithm proceeds as below. It utilizes the proximal operators of $g^*$ and $f$, resembling the primal-dual formulation. The algorithm is included in appendix \ref{sub:pseudo}.

\subsubsection{$l_1$ vs $l_2$ and Early Stopping}

We found that $l_1$ almost always performs better than $l_2$ (\ref{heatmaps} \ref{gridsearchplots}). Also, $l_2$ displays much more sensitivity to outliers. This can be explained by the squaring of the terms and the facts that n-norms tend to augment their largest elements disproportionately  as they n increases.

Moreover, the $l_1$ norm is known to induce sparsity. It therefore induces more clearer edges as approximately close values become the same, while the $l_2$ norm as a penalty does not clips values to 0, or in this case to say $b$.  In some sense, when we implement the proximal operators, the problem "$l_1$" or "$l_2$" sort of becomes the penalty, and the $\frac{1}{2} \| y - u \|_2^2$ sort of becomes our actual function. In addition, because we are in a non-blind framework, we have knowledge of the blurring kernel $K$ and therefore know of the map say $x \mapsto \|K x - b\|_1$. Clipping to $0$ when $u = b$ as shown in the derivation of the proximal operator of the $l_1$ norm reminds the sparsity-inducing feature of this map. It naturally leads to clearer edges. As an example, see figure (\ref{fig:l1vl2}).

We are also implementing a form of early stopping based on a moving average. That is, we are keeping a history of the previous \texttt{i.malength} number of previous objective value, then taking the sample average, and stopping if the current objective value is greater than that of the sample mean.



\subsection{Conclusion}

We experienced a few problems when completing the project, which first and foremost included learning the programming language. With our group having little to no experience with Matlab, we had to familiarize ourselves with the syntax of the language. With syntax resembling a mix of python and R, and plenty of helpful functions with insightful documentation/instructions, we were quickly able to adjust. We had issues that had us scratching our heads caused by not using $clearvals$. Another task that posed a challenge was parameter tuning. With four algorithms to adjust and 2 norms to choose from in each algorithm, we had to spend sometime optimizing parameters. Implementing heat maps made it easier to find the parameters for the respective algorithms. 

The winner of the algorithms would be between the DRP and ADMM algorithms. They both worked quickly and had very nice results. In terms of which algorithm was easier to code, it is tricky to say, as we had very little experience with Matlab when coding the first algorithm, DRP, but we built up experience in time for writing ADMM. However, DRP algorithm is objectively easier to understand and code when comparing it to the implementation of the ADMM algorithm. With that being said, we would recommend the DRP algorithm to an individual with less experience with Matlab and convex optimization, and recommend the ADMM algorithm the seasoned veterans of the subject.

\subsection{Pseudocode of the Algorithms}%
\label{sub:pseudo}
\begin{algorithm}[H]
  \label{algo:PRS}
 \SetKw{Null}{NULL}
 \SetKw{Return}{return}
 \Input{ Blurred image $b$, step size $t$, relaxation parameter $\rho$}
 \Output{ Deblurred image $x$ }
\Initialize{ $(z^0_1, z^0_2)$ }

\For { $k = 1, 2, \hdots, $ } {
  $x^k \leftarrow \prox_{tf}(z^{k-1}_1)$ \tcp*{Performs resolvent of $\mathcal{A}$} 
  $y^k \leftarrow \prox_{tg}(z^{k-1}_2)$ \;
  $u^k \leftarrow (I+A^TA)^{-1} \big ( 2x^k-z^{k-1}_1 +
  A^T(2y^k-z^{k-1}_2) \big )$ \tcp*{Performs
    resolvent of $\mathcal{B}$}
  $v^k \leftarrow A(u^k)$ \;
  $z^k_1 \leftarrow z^{k-1}_1 + \rho(u^k-x^k)$ \;
  $z^k_2 \leftarrow z^{k-1}_2 + \rho(v^k-y^k)$ \;
} 
\Return $x\text{Sol} = \prox_{tf}(z^k)$\;
\caption{Primal Douglas-Rachford Splitting}\label{alg: DRprimal} 
\end{algorithm}
\begin{algorithm}[h!]
  \label{algo:PDRS}
  \SetKw{Null}{NULL}
  \SetKw{Return}{return}
  \Input{ Blurred image $b$, step size $t$, relaxation parameter $\rho$}
  \Output{ Deblurred image $x$ }
 \Initialize{ $p^0, q^0$ }
 
 \For { $k = 1, 2, \hdots, $ } {
   $x^k \leftarrow \prox_{tf}(p^{k-1}_1)$ \tcp*{Performs resolvent of $\mathcal{A}$} 
   $z^k \leftarrow \prox_{tg^*}(q^{k-1})$ \;
   $\begin{bmatrix} w^k\\ v^k  \end{bmatrix} = \begin{bmatrix} I &
    t A^T\\ -t A & I \end{bmatrix}^{-1} \begin{bmatrix}
     2x^k-p^{k-1}\\ 2z^k-q^{k-1} \end{bmatrix} $ \tcp*{Performs resolvent of $\mathcal{B}$} 
   $p^k \leftarrow p^{k-1} + \rho(w^k-x^k)$ \;
   $q^k \leftarrow q^{k-1} + \rho(v^k-z^k)$ \;
 } 
 \Return $x\text{Sol} = \prox_{tf}(p^k)$\;
 \caption{Primal-Dual Douglas-Rachford Splitting}\label{alg: DRpd} 
 \end{algorithm}

\begin{algorithm}[h!]
  \label{algo:ADMM}
 \SetKw{Null}{NULL}
 \SetKw{Return}{return}
 \Input{ Blurred image $b$, step size $t$, relaxation parameter $\rho$}
 \Output{ Deblurred image $x$ }
\Initialize{ $x^0, u^0, y^0, w^0, z^0$ }

\For { $k = 1, 2, \hdots, $ } {
  $ x^k \leftarrow (I+A^TA)^{-1} \left( u^{k-1} + A^T y^{k-1} - \frac{1}{t} (w^{k-1} + A^T z^{k-1} )\right) $ \;
  $ u^k \leftarrow \text{prox}_{t^{-1}f}(\rho x^k + (1-\rho)u^{k-1} +  w^{k-1}/t )$ \;
  $ y^k \leftarrow \text{prox}_{t^{-1}g}(\rho Ax^k + (1-\rho)y^{k-1} + z^{k-1}/t )$ \;
  $ w^k \leftarrow w^{k-1} + t(x^k - u^k)$ \;
  $ z^k \leftarrow z^{k-1} + t(Ax^k - y^k)$ \;
} 
\Return $x\text{Sol} =  (I+A^TA)^{-1} \left( u^{k} + A^T y^{k} - \frac{1}{t} (w^{k} + A^T z^{k} )\right) $\;
\caption{Alternating Direction Method of Multipliers (ADMM)}\label{alg: ADMM} 
\end{algorithm}
\begin{algorithm}[h!]
    \label{algo:ChambollePock}
 \SetKw{Null}{NULL}
 \SetKw{Return}{return}
 \Input{ Blurred image $b$, step sizes $t$, $s$}
 \Output{ Deblurred image $x$ }
\Initialize{ $x^0, y^0, z^0$ }

\For { $k = 1, 2, \hdots, $ } {
  $ y^k = \text{prox}_{sg^{*}}(y^{k - 1} + sAz^{k-1})$    \;
  $ x^k = \text{prox}_{tf}(x^{k-1} - tA^{T}y^{k})$ \;
  $	z^{k}  = 2x^{k}  - x^{k-1} $ \;
} 
\Return $x\text{Sol} = x^{k}$\;
\caption{Chambolle-Pock Algorithm}\label{alg:ChambollePock} 
\end{algorithm}
